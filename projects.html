Code Sample: Dynamic UI Modal Generator

Example snippet illustrating custom HTML5/CSS3 modal dialog creation and asynchronous backend communication via Google Apps Script.

function openNewCase() {
  var html = HtmlService.createHtmlOutput(
    '<!DOCTYPE html><html><head><base target="_top"><style>' +
    'body { font-family: -apple-system, sans-serif; padding: 16px; font-size: 13px; }' +
    '.form-group { margin-bottom: 14px; }' +
    'input, select, textarea { width: 100%; padding: 8px; border: 1px solid #dadce0; border-radius: 4px; }' +
    '</style></head><body>' +
    '<h3>New Ticket Intake</h3>' +
    '<p>System modal initialized successfully.</p>' +
    '</body></html>'
  ).setWidth(488).setHeight(300);
  
  SpreadsheetApp.getUi().showModalDialog(html, "Create New Case");
}
